6)--IF-THEN-ELSIF-ELSE.

DECLARE 
       AGE NUMBER(10):=:AGE;
BEGIN
     IF(AGE > 18) THEN 
         DBMS_OUTPUT.PUT_LINE('TICKET PRICE 200');
     ELSIF(AGE < 10)THEN 
         DBMS_OUTPUT.PUT_LINE('TICKET IS PRICE 100:');
     ELSIF(AGE > 30) THEN
         DBMS_OUTPUT.PUT_LINE('TICKTE PRICE IS 300:');
     ELSE
         DBMS_OUTPUT.PUT_LINE('TICKET  REGULAR PRICE IS 150:');
     END IF;
END;








7)--Input three no and find out maximum and minimum from it.

DECLARE 
       A NUMBER(10):=:A;
       B NUMBER(10):=:B;
       C NUMBER(10):=:C;
BEGIN
     IF (A<B AND B>C) THEN
               DBMS_OUTPUT.PUT_LINE('MAX='||B);
      
     ELSIF(B<C AND C>A) THEN 
            DBMS_OUTPUT.PUT_LINE('MAX:='||C);
        
     ELSE
            DBMS_OUTPUT.PUT_LINE('MAX:='||A);
         
     END IF;
       
       IF (A>B AND B<C) THEN
               DBMS_OUTPUT.PUT_LINE('MIN='||B);
      
     ELSIF(B>C AND C<A) THEN 
            DBMS_OUTPUT.PUT_LINE('MIN:='||C);
        
     ELSE
            DBMS_OUTPUT.PUT_LINE('MIN:='||A);
         
     END IF;
END ;



(9)--PL/SQL Case Statement. 
DECLARE 
       DEPT CHAR(1):='B';
BEGIN
     CASE DEPT 
           WHEN 'A' THEN
              DBMS_OUTPUT.PUT_LINE('BCA DEPARTMENT');
           WHEN 'B' THEN 
              DBMS_OUTPUT.PUT_LINE('MCA DEPARTMENT');
           WHEN 'M' THEN 
              DBMS_OUTPUT.PUT_LINE('CA DEPARTMENT');
     ELSE 
         DBMS_OUTPUT.PUT_LINE('INVALID DEPARTMENT');
     END CASE;
END;
      



(10)-- print first 10 numbers using basic loop.. exit loop.
DECLARE 
       I NUMBER(3):=1;
BEGIN
      LOOP
          DBMS_OUTPUT.PUT_LINE(I);
      EXIT WHEN (I = 10);
      I:=I+1;
      END LOOP;
END ;

(11)-- print first 10 numbers using while loop.
DECLARE 
       I NUMBER(10):=1;

BEGIN 
     WHILE (I <= 10)
     LOOP
      DBMS_OUTPUT.PUT_LINE(I);
     I:=I+1;
     END LOOP;
END;
