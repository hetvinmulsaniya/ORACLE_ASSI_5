--IF-THEN-ELSIF-ELSE.

DECLARE 
       AGE NUMBER(10):=:AGE;
BEGIN
     IF(AGE > 18) THEN 
         DBMS_OUTPUT.PUT_LINE('TICKET PRICE 200');
     ELSIF(AGE < 10)THEN 
         DBMS_OUTPUT.PUT_LINE('TICKET IS PRICE 100:');
     ELSIF(AGE > 30) THEN
         DBMS_OUTPUT.PUT_LINE('TICKTE PRICE IS 300:');
     ELSE
         DBMS_OUTPUT.PUT_LINE('TICKET  REGULAR PRICE IS 150:');
     END IF;
END;








--Input three no and find out maximum and minimum from it.

DECLARE 
       A NUMBER(10):=:A;
       B NUMBER(10):=:B;
       C NUMBER(10):=:C;
BEGIN
     IF (A<B AND A<C) THEN
        IF(A < B ) THEN
             DBMS_OUTPUT.PUT_LINE('MAX:'||A);
        ELSE
             DBMS_OUTPUT.PUT_LINE('MIN');
     ELSIF(B<C AND B<A) THEN 
        IF(B < C ) THEN
             DBMS_OUTPUT.PUT_LINE('MAX:');
        ELSE
             DBMS_OUTPUT.PUT_LINE('MIN');

     ELSIF(C<B AND C<A) THEN 
         IF (C < B )THEN 
              DBMS_OUTPUT.PUT_LINE('MAX:');
         ELSE 
              DBMS_OUTPUT.PUT_LINE('MIN');
     END IF;
END ;
